cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rmtc_bot_core LANGUAGES CXX)
  include(CTest)
endif()
get_filename_component(BOT_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
option(RMTC_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(RMTC_OPTIMIZATION "O2" CACHE STRING "Research Release optimization: O2 or O3")
set_property(CACHE RMTC_OPTIMIZATION PROPERTY STRINGS O2 O3)
option(RMTC_LTO "Enable interprocedural optimization for Release research builds" OFF)
if(NOT RMTC_OPTIMIZATION MATCHES "^O[23]$")
  message(FATAL_ERROR "RMTC_OPTIMIZATION must be O2 or O3")
endif()
find_package(nlohmann_json 3.10 QUIET)
if(NOT nlohmann_json_FOUND)
  if(NOT EXISTS "${BOT_ROOT}/vendor/include/nlohmann/json.hpp")
    message(FATAL_ERROR "nlohmann/json missing; restore the bundled v3.11.3 header")
  endif()
  add_library(rmtc_json INTERFACE)
  target_include_directories(rmtc_json SYSTEM INTERFACE "${BOT_ROOT}/vendor/include")
  add_library(nlohmann_json::nlohmann_json ALIAS rmtc_json)
endif()
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(bot_core STATIC "${BOT_ROOT}/src/protocol.cpp" "${BOT_ROOT}/src/core.cpp" "${BOT_ROOT}/src/respawn.cpp" "${BOT_ROOT}/src/combat_ledger.cpp"
    "${BOT_ROOT}/src/events.cpp" "${BOT_ROOT}/src/llm.cpp" "${BOT_ROOT}/src/event_memory.cpp" "${BOT_ROOT}/src/battle.cpp"
    "${BOT_ROOT}/src/config.cpp" "${BOT_ROOT}/src/runtime.cpp" "${BOT_ROOT}/src/escape.cpp" "${BOT_ROOT}/src/opportunity.cpp" "${BOT_ROOT}/src/objective.cpp" "${BOT_ROOT}/src/recon.cpp" "${BOT_ROOT}/src/pursuit.cpp" "${BOT_ROOT}/src/prompt.cpp")
target_compile_features(bot_core PUBLIC cxx_std_17)
target_compile_options(bot_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(bot_core PUBLIC "$<$<CONFIG:Release>:-${RMTC_OPTIMIZATION}>")
target_include_directories(bot_core PUBLIC "${BOT_ROOT}/include")
target_link_libraries(bot_core PUBLIC nlohmann_json::nlohmann_json CURL::libcurl Threads::Threads)
if(RMTC_SANITIZE)
  if(CMAKE_BUILD_TYPE STREQUAL "Release" OR RMTC_LTO)
    message(FATAL_ERROR "Use Debug without LTO for sanitizer checks")
  endif()
  target_compile_options(bot_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  target_link_options(bot_core PUBLIC -fsanitize=address,undefined)
endif()
if(RMTC_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_ok OUTPUT ipo_error)
  if(NOT ipo_ok)
    message(FATAL_ERROR "LTO unsupported: ${ipo_error}")
  endif()
  set_property(TARGET bot_core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
add_executable(replay_tool "${BOT_ROOT}/tools/replay_main.cpp")
target_link_libraries(replay_tool PRIVATE bot_core)
target_compile_options(replay_tool PRIVATE -Wall -Wextra -Wpedantic)
add_executable(event_eval_tool "${BOT_ROOT}/tools/event_eval_tool.cpp")
target_link_libraries(event_eval_tool PRIVATE bot_core)
target_compile_options(event_eval_tool PRIVATE -Wall -Wextra -Wpedantic)
