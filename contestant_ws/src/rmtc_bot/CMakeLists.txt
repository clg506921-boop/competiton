cmake_minimum_required(VERSION 3.16)
project(rmtc_bot LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Threads REQUIRED)
include(CTest)
add_subdirectory(core)
add_executable(turing_bot_node src/ros_node.cpp)
target_link_libraries(turing_bot_node bot_core Threads::Threads)
target_compile_options(turing_bot_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(turing_bot_node rclcpp std_msgs)
install(TARGETS turing_bot_node replay_tool DESTINATION lib/${PROJECT_NAME})
ament_package()
